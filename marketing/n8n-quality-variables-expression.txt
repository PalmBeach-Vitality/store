const preferred = ["ultra detailed", "extremely detailed", "hyper-detailed", "razor sharp focus", "tack sharp", "crystal clear", "ultra sharp", "8k resolution", "photorealistic", "hyperrealistic", "ultra realistic", "cinematic lighting", "dramatic lighting", "volumetric lighting", "soft diffused lighting", "studio lighting", "rim lighting", "high dynamic range", "HDR", "rich contrast", "perfect exposure", "shallow depth of field", "creamy bokeh", "perfect composition", "elegant framing", "premium aesthetic", "polished look", "clean and crisp", "refined details", "high fidelity", "realistic textures", "intricate textures", "detailed surface texture", "global illumination", "ambient occlusion", "soft shadows", "realistic reflections", "high-end commercial look", "editorial photography style", "medium format look", "DSLR sharpness", "professional photography"];

function pick(n) {
  const copy = [...preferred];
  const out = [];
  for (let i = 0; i < n && copy.length; i++) {
    const idx = Math.floor(Math.random() * copy.length);
    out.push(copy.splice(idx, 1)[0]);
  }
  return out;
}

const quality_vars = pick(6);
const quality_suffix = quality_vars.join(', ');

return [{
  json: {
    ...$json,
    quality_vars,
    quality_suffix,
  }
}];
